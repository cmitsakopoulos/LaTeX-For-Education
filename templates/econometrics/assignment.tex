\begin{problembox}[50 Points]{1: Two-Way Fixed Effects Demeaning and Algebraic Equivalence}
Consider the standard panel regression model with unit fixed effects $\alpha_i$ and time fixed effects $\lambda_t$:
\begin{equation*}
    Y_{it} = \alpha_i + \lambda_t + \delta D_{it} + \varepsilon_{it}, \quad i = 1, \dots, N, \;\; t = 1, \dots, T
\end{equation*}
\begin{enumerate}[label=(\alph*)]
    \item Define the within-group double demeaning transformation $\ddot{Y}_{it}$ and prove that $\ddot{\alpha}_i = 0$ and $\ddot{\lambda}_t = 0$.
    \item Derive the OLS estimator of $\delta$ applied to the double-demeaned equation.
\end{enumerate}
\end{problembox}

\ifsolutions
\begin{solutionbox}
\textbf{Part (a):} Define $\bar{Y}_i = \frac{1}{T}\sum_{t=1}^T Y_{it}$, $\bar{Y}_t = \frac{1}{N}\sum_{i=1}^N Y_{it}$, and $\bar{Y} = \frac{1}{NT}\sum_{i=1}^N\sum_{t=1}^T Y_{it}$.
The double-demeaned variable is:
\begin{equation*}
    \ddot{Y}_{it} = Y_{it} - \bar{Y}_i - \bar{Y}_t + \bar{Y}
\end{equation*}
For the unit effect $\alpha_i$:
$\bar{\alpha}_i = \alpha_i$, $\bar{\alpha}_t = \frac{1}{N}\sum_i \alpha_i = \bar{\alpha}$, and $\bar{\alpha} = \bar{\alpha}$.
Hence:
\begin{equation*}
    \ddot{\alpha}_i = \alpha_i - \alpha_i - \bar{\alpha} + \bar{\alpha} = 0
\end{equation*}
Similarly, $\ddot{\lambda}_t = \lambda_t - \bar{\lambda} - \lambda_t + \bar{\lambda} = 0$.

\textbf{Part (b):} The transformed model is $\ddot{Y}_{it} = \delta \ddot{D}_{it} + \ddot{\varepsilon}_{it}$.
The within estimator is:
\begin{equation*}
    \hat{\delta}_{\mathrm{FE}} = \frac{\sum_{i=1}^N \sum_{t=1}^T \ddot{D}_{it} \ddot{Y}_{it}}{\sum_{i=1}^N \sum_{t=1}^T \ddot{D}_{it}^2} \quad \blacksquare
\end{equation*}
\end{solutionbox}
\fi

\begin{problembox}[50 Points]{2: Equivalence of $2 \times 2$ DiD and Sample Means}
In a $2$-period, $2$-group setting ($t \in \{1, 2\}$, $D \in \{0, 1\}$), where treatment occurs at $t=2$ for the treated group:
\begin{enumerate}[label=(\alph*)]
    \item State the regression specification $Y_{it} = \beta_0 + \beta_1 \mathrm{Treat}_i + \beta_2 \mathrm{Post}_t + \delta (\mathrm{Treat}_i \times \mathrm{Post}_t) + \varepsilon_{it}$.
    \item Show that the OLS estimate $\hat{\delta}$ equals $(\bar{Y}_{T,2} - \bar{Y}_{T,1}) - (\bar{Y}_{C,2} - \bar{Y}_{C,1})$.
\end{enumerate}
\end{problembox}

\ifsolutions
\begin{solutionbox}
Evaluating the conditional expectations of $Y$ for the four cells:
\begin{align*}
    \mathbb{E}[Y \mid \mathrm{Treat}=0, \mathrm{Post}=0] &= \beta_0 = \bar{Y}_{C,1} \\
    \mathbb{E}[Y \mid \mathrm{Treat}=0, \mathrm{Post}=1] &= \beta_0 + \beta_2 = \bar{Y}_{C,2} \implies \beta_2 = \bar{Y}_{C,2} - \bar{Y}_{C,1} \\
    \mathbb{E}[Y \mid \mathrm{Treat}=1, \mathrm{Post}=0] &= \beta_0 + \beta_1 = \bar{Y}_{T,1} \implies \beta_1 = \bar{Y}_{T,1} - \bar{Y}_{C,1} \\
    \mathbb{E}[Y \mid \mathrm{Treat}=1, \mathrm{Post}=1] &= \beta_0 + \beta_1 + \beta_2 + \delta = \bar{Y}_{T,2}
\end{align*}
Solving for $\delta$:
\begin{equation*}
    \delta = \bar{Y}_{T,2} - (\beta_0 + \beta_1 + \beta_2) = (\bar{Y}_{T,2} - \bar{Y}_{T,1}) - (\bar{Y}_{C,2} - \bar{Y}_{C,1}) \quad \blacksquare
\end{equation*}
\end{solutionbox}
\fi
