\begin{problembox}[50 Points]{1: Moment Generating Functions and the Central Limit Theorem}
Let $X_1, X_2, \dots, X_n$ be a sequence of independent and identically distributed (i.i.d.) random variables with mean $\mu = 0$, variance $\sigma^2 = 1$, and moment generating function $M_X(t) = \mathbb{E}[e^{tX}]$ defined in a neighborhood of $t=0$.
\begin{enumerate}[label=(\alph*)]
    \item Write the Taylor expansion of $M_X(t)$ around $t=0$ up to second order.
    \item Define the standardized sum $Z_n = \frac{1}{\sqrt{n}} \sum_{i=1}^n X_i$. Find the MGF $M_{Z_n}(t)$ in terms of $M_X(t)$.
    \item Prove that $\lim_{n \to \infty} M_{Z_n}(t) = e^{t^2/2}$, thereby establishing the Central Limit Theorem.
\end{enumerate}
\end{problembox}

\ifsolutions
\begin{solutionbox}
\textbf{Part (a):} Expanding $M_X(t) = \mathbb{E}[1 + tX + \frac{t^2 X^2}{2} + o(t^2)]$:
Since $\mathbb{E}[X] = 0$ and $\mathbb{E}[X^2] = 1$, we obtain:
\begin{equation*}
    M_X(t) = 1 + \frac{t^2}{2} + o(t^2)
\end{equation*}

\textbf{Part (b):} By independence of $X_i$:
\begin{equation*}
    M_{Z_n}(t) = \mathbb{E}\left[\exp\left(\frac{t}{\sqrt{n}} \sum_{i=1}^n X_i\right)\right] = \prod_{i=1}^n M_X\left(\frac{t}{\sqrt{n}}\right) = \left[ M_X\left(\frac{t}{\sqrt{n}}\right) \right]^n
\end{equation*}

\textbf{Part (c):} Substituting the Taylor expansion:
\begin{equation*}
    M_{Z_n}(t) = \left[ 1 + \frac{t^2}{2n} + o\left(\frac{t^2}{n}\right) \right]^n
\end{equation*}
Using the fundamental limit $\lim_{n \to \infty} (1 + \frac{a}{n})^n = e^a$, we set $a = t^2/2$:
\begin{equation*}
    \lim_{n \to \infty} M_{Z_n}(t) = \exp\left(\frac{t^2}{2}\right)
\end{equation*}
Since $e^{t^2/2}$ is the moment generating function of a standard normal distribution $\mathcal{N}(0, 1)$, by the Continuity Theorem for MGFs, $Z_n \xrightarrow{d} \mathcal{N}(0, 1)$. \hfill $\blacksquare$
\end{solutionbox}
\fi

\begin{problembox}[50 Points]{2: Asymptotic Variance of the Log Odds Ratio}
For a $2 \times 2$ contingency table with independent multinomial counts $(n_{11}, n_{12}, n_{21}, n_{22})$, the sample Odds Ratio is $\widehat{\mathrm{OR}} = \frac{n_{11} n_{22}}{n_{12} n_{21}}$.
\begin{enumerate}[label=(\alph*)]
    \item Apply the Delta Method to derive the asymptotic variance of $\ln \widehat{\mathrm{OR}}$.
    \item Derive the formula for a $(1-\alpha)$ confidence interval for the true population Odds Ratio.
\end{enumerate}
\end{problembox}

\ifsolutions
\begin{solutionbox}
\textbf{Part (a):} Let $\theta = \ln \widehat{\mathrm{OR}} = \ln n_{11} - \ln n_{12} - \ln n_{21} + \ln n_{22}$.
Using the delta method for independent Poisson/multinomial components:
\begin{equation*}
    \mathrm{Var}(\ln \widehat{\mathrm{OR}}) \approx \sum_{i,j} \left(\frac{\partial \theta}{\partial n_{ij}}\right)^2 \mathrm{Var}(n_{ij}) = \frac{1}{n_{11}} + \frac{1}{n_{12}} + \frac{1}{n_{21}} + \frac{1}{n_{22}}
\end{equation*}

\textbf{Part (b):} The standard error is $\mathrm{SE}(\ln \widehat{\mathrm{OR}}) = \sqrt{\frac{1}{n_{11}} + \frac{1}{n_{12}} + \frac{1}{n_{21}} + \frac{1}{n_{22}}}$.
The $(1-\alpha)$ confidence interval for $\mathrm{OR}$ is:
\begin{equation*}
    \left[ \exp\left(\ln \widehat{\mathrm{OR}} - z_{1-\alpha/2} \mathrm{SE}\right), \;\; \exp\left(\ln \widehat{\mathrm{OR}} + z_{1-\alpha/2} \mathrm{SE}\right) \right] \quad \blacksquare
\end{equation*}
\end{solutionbox}
\fi
